Estrategia

// 1. El Morphing del Canvas base de la web a Tono Bodak (Arena/Clay)
ScrollTrigger.create({
  trigger: "#section-spread",
  start: "top 50%", 
  end: "bottom 0%",
  onEnter: () => gsap.to("body", { backgroundColor: "var(--bg-morph)", color: "var(--txt-morph)" }),
  onLeaveBack: () => gsap.to("body", { backgroundColor: "var(--bg)", color: "var(--txt)" }) // Vuelve oscuro
});

// 2. Dispersión: Las tarjetas laterales se repelen hacia fuera
// Tarjeta Izquierda (T1)
gsap.to(".card-1", {
  xPercent: -120, rotation: -12, scale: 0.9, 
  scrollTrigger: { trigger: ".spread-wrap", start: "top 60%", end: "bottom 80%", scrub: 1 }
});
// Tarjeta Derecha (T3)
gsap.to(".card-3", {
  xPercent: 120, rotation: 12, scale: 0.9, 
  scrollTrigger: { trigger: ".spread-wrap", start: "top 60%", end: "bottom 80%", scrub: 1 }
});
